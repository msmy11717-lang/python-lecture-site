import csv

def load_csv_as_text(filename):
    rows = []
    with open(filename, "r", encoding="utf-8-sig") as f:
        reader = csv.reader(f)
        for row in reader:
            rows.append(",".join(row))
    return "\n".join(rows)

csv_text = load_csv_as_text("data.csv")

以下のデータを分析してください：
